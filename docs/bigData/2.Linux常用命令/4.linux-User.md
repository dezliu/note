```
useradd liudezhi
#添加普通用户
passwd liudezhi
#只有root用户可以拿这个命令去修改别人的密码，普通用户只能用来修改自己的密码
id liudezhi

```

**两个用户之间共享数据**

```
useradd user01
useradd user02
passwd user01
passwd user02
cd /
mkdir share
groupadd usershare
usermod -a -G usershare user01              #将user01用户添加到usershare组
usermod -a -G usershare user02              #将user01用户添加到usershare组
chown root:usershare share                     #为share文件夹添加usershare组
# chmod ugo:(user group other) +/- or 777
chmod g+w share
chmod o-rx share
# user01登录
cd /share
vim a.txt
# user02登录
cd /share
vim a.txt
# user02只能读取文件，不能修改
ls -l a.txt
$ -rw-rw-r-- 1 user01 user01 8 Nov 22 19:37 a.txt
# 两种方法：1-修改文件权限
chmod 666 a.txt
# 2-

```

**eg: mysql服务器**

```
# 一般放在/opt下
# mysql中一般有bin和sbin目录，bin中一般放的是客户端和使用的命令
# sbin目录中一般放的是服务命令(exp:start stop)
useradd mysqladmin
chown -R mysqladmin:root mysql
groupadd mysqlgroup
useradd mysql01
useradd mysql02
usermod -a -G mysqlGroup mysql01
usermod -a -G mysqlGroup mysql02
chown -R mysqladmin:mysqlgroup mysql
cd mysql
# ...

```
