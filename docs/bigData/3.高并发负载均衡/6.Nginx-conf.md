/opt/liudezhi/nginx/conf/nginx.conf

```
#user nobody               # 用户
worker_processes 1          # work进程数(硬件服务器计有多少颗物理CPU核心，这里的数字写成几，或者它的两倍)
# nginx是一种nio（非阻塞）的方式，会用一个进程接收很多个连接
# error_log ...   # 日志存储位置
events {                        # nginx支持热加载热升级，不用停服务
    worker_connections 1024; # worker可以处理的客户端连接数，官方说法，可以处理万级别，但是直接修改不能生效。因为linux默认一个进程只可以生成1024个文件描述符，每个socket都是一个文件描述符
}
# ps -ef | grep nginx  有 master和worker两个进程
# master 进程不处理客户端连接的事情，只有worker进程处理客户端的连接
# worker的用户是root，worker是nobody
# worker是master的子进程
# 当配置文件修改时，只需要reload，worker进程连接的客户端继续处理，会孵化出新的work进程读取新的配置文件，接受新的请求，当老的workder上所有的连接处理完没有了，把这个worker回收。
http {
    #log_format main...  # 日志记录的字段及顺序，看日志不知道在些什么的时候可以看日志里的字段。
    sendfile on;            # 零拷贝
    #tcp_nopush on;     # 缓冲区开关
    #keepalive_timeout 0； # http协议中字段名，防止频繁三次握手四次分手，当客户端持续多长时间不发请求了才四次分手。在生成系统中一定是开启的，但是实验中为了看到切换的效果把它设置为0
    keepalive_timeout 65;
    #gzip on;           # 开启压缩，好处：I/O传输更快，可以传输更多东西。
    server {
        # nginx支持虚拟服务器的概念，
        location / {
            root html;          # 目录，指定页面文件位置 /opt/liudezhi/nginx/html，一个server只能有一个root
            # eg:访问192.168.231.31:80/abc/a.html 是基于root目录 /html/abc/a.html
            index index.html index.htm; # 访问80端口的时候，两个页面有哪个返回哪个
        }
    }
}
#因为支持虚拟服务器，可以开启多个80端口的服务器
# 添加：
server {
    listen 80;
    server_name www.liudezhi.com;
    location / {
        root /mnt;
        autoindex on;
    }
}
# 原本的server修改
server_name www.xiashuang.com

# windows hosts文件，
# 192.168.231.31 www.liudezhi.com www.xiashuang.com
# 然后通过浏览器访问www.liudezhi.com和www.xiashuang.com即可进入不同页面
# 请求头中会带有Host字段，如果用ip地址访问这个字段是ip地址
# nginx收到请求头，判定ip，port，host，决定server
# nginx location匹配：用客户端的uri匹配location的uri
# 反向代理是基于location匹配之后

http、server、location都属于webserver基本的逻辑，所以它属于ngx_http_core_module模块

```
location = / {
    [ configuration A ]
}

location / {
    [ configuration B ]
}

location /documents/ {
    [ configuration C ]
}

location ^~ /images/ {
    [ configuration D ]
}

location ~* \.(gif|jpg|jpeg)$ {
    [ configuration E ]
}
# 请求“/”匹配配置A， 请求“/index.html”匹配配置B， 请求“/documents/document.html”匹配配置C， 请求“/images/1.gif”匹配配置D， 请求“/documents/1.jpg”匹配配置E。

```

##### location配置规则

- location的执行逻辑跟location的编辑顺序无关。
 矫正：这句话不全对，“普通location“的匹配规则是”最大前缀”，因此“普通location”的确与location的编辑顺序无关。

- 但是“正则 location”的匹配规则是"顺序匹配，且只要匹配到第一个就停止后面的匹配"

- “普通location”与“正则location”之间的匹配顺序是 先匹配普通location再“考虑"匹配正则location

- 注意这里的“考虑”是“可能”对的意思，也就是说匹配完“普通 location”后，有的时候需要继续匹配“正则location”，有的时候则不需要继续匹配“正则location”。两种情况下，不需要继续匹配正则location：

  - 当普通location前面指定了 "^~"，特别告诉Nginx本条普通location一旦匹配上，则不需要继续正则匹配；

  - 当普通location恰好严格匹配上，不是最大前缀匹配，则不再继续匹配正则

```
server {
    #添加一个location
    location /liudezhi {
        proxy_pass http://192.168.231.32:80/;           # 反向代理
    }
    location /xiashuang {
        proxy_pass http://www.baidu.com/;           # 反向代理
    }
}
# service nginx reload
# 访问www.liudezhi.com/liudezhi即可代理到指定的192.168.231.32
# www.baidu.com要用到域名解析，若下一跳地址没有会报错
# 会发生客户端跳转（百度会让从http跳转到https）
# 将http改成https即可（报错）

location ~* ^/s {
    proxy_pass http://www.baidu.com;
}
location /liu {
    proxy_pass http://www.baidu.com/;
}

```

```
# 再server{}上添加，定义一个服务器池和server同级
upstream liu {
    server 192.168.231.32:80;

server {
    location /liudezhi {
        proxy_pass http://liu/;
    }
