1. 装饰模式

1. 代理模式

  - 概念：

    1. 真实对象：被代理的对象

    1. 代理对象

    1. 代理模式：代理对象代理真实对象，达到增强真实对象功能的目的

  - 实现方式

    1. 静态代理：有一个类文件描述代理模式

    1. 动态代理：在内存中形成代理类

      - 实现步骤：

        1. 代理对象和真实对象实现相同的接口

        1. 代理对象 = Proxy.newProxyInstance();

        1. 使用代理对象调用方法

        1. 增强方法

      - 增强方式：

        1. 增强参数列表

        1. 增强返回值类型

        1. 增强方法体执行逻辑

package cn.liudezhi.proxy;

import java.lang.reflect.InvocationHandler;
import java.lang.reflect.Method;
import java.lang.reflect.Proxy;

public class ProxyTest {
    public static void main(String[] args) {
        // 1. 创建真实对象
        Lenovo lenovo = new Lenovo();
        // 2. 动态代理增强lenovo 对象
        /*
            三个参数：
                1. 类加载器：真实对下个你.getClass().getClassLoader()
                2. 接口数数组：真实对象.getClass().getInterfaces()
                3. 处理器：new InvocationHandler()
            代理对象和真实对象实现相同的接口，因此返回的对象，可以直接转为接口的类型
         */
        SaleComputer proxy_lenovo = (SaleComputer)Proxy.newProxyInstance(lenovo.getClass().getClassLoader(), lenovo.getClass().getInterfaces(), new InvocationHandler() {
            /*
                代理逻辑编写的方法：代理对象调用的所有方法都会触发该方法执行
                    参数：
                        1. proxy：代理对象
                        2. method：代理对象调用的方法，被封装为的对象
                        3. args：代理对象调用方法时，传递的实际参数
             */
            @Override
            public Object invoke(Object proxy, Method method, Object[] args) throws Throwable {
//                System.out.println("该方法执行了...");
//                System.out.println(method.getName());
//                System.out.println(args[0]);

                // 判断是否是sale 方法
                if(method.getName().equals("sale")) {
                    // 1. 增强参数
                    double money = (double) args[0];
                    money = money * 0.85;
                    System.out.println("转车接...");           // 方法体逻辑增强
                    // 使用真实对象调用该方法
                    String obj = (String) method.invoke(lenovo, money);
                    System.out.println("免费送货...");          // 方法体逻辑增强
                    // 2. 增强返回值
                    return obj + "_鼠标垫";
                } else {
                    // 使用真实对象调用该方法
                    Object obj = method.invoke(lenovo, args);
                    return obj;
                }
            }
        });
        // 2. 调用方法
        String computer = proxy_lenovo.sale(8000);
        System.out.println(computer);
        proxy_lenovo.show();
    }
}
