<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <meta http-equiv="X-UA-Compatible" content="ie=edge">
    <title>Document</title>
    <script src="./lib/vue.js"></script>
</head>
<body>
    <div id="app">
        <input type="button" value="start" @click="startLamp">
        <input type="button" value="stop" @click="endLamp">
        <h4>{{ msg }}</h4>
    </div>
    <script>
        // 注意：在VM实例中，如果想要获取data 上的数据，或者想要调用methods 中的方法，必须通过thiss.数据属性名 或 this.方法名 来进行访问
        // 这里的this 就是我们new出来额VM实例对象
        var vm = new Vue({
            el: '#app',
            data: {
                msg: 'liudezhi',
                intervalId: null    // 在data上定义定时器id
            },
            methods: {
                startLamp: function() {
                    if(this.intervalId == null) {       // 防止多次点击开启多个定时器
                            this.intervalId = setInterval(() => {     // 箭头函数，内部this指向外部this，如果是function() {} 要在外部配置var self = this; 里面用self 代替this(箭头函数内部的this永远和外部的this保持一致)
                            var start = this.msg.substring(0, 1);       // 获取msg的第一个字符
                            var end = this.msg.substring(1);            // 截取除了第一个意外的字符
                            // 拼接得到新的字符串，并赋值给 this.msg
                            this.msg = end + start;     // 注意：VM实例，会监听自己身上 data中所有数据的改变，只要数据易发生改变，就会自动把最新的数据，从data上同步到页面中去;[好处: 程序员只需要关心数据，不需要考虑如何重新渲染DOM页面]
                        }, 400);
                    }
                },
                endLamp() {         //ES6写法,停止定时器
                    clearInterval(this.intervalId);
                    this.intervalId = null;     // 每当清除了定时器，需要重新把intervalId 置为null
                }
            }
        });
    </script>
</body>
