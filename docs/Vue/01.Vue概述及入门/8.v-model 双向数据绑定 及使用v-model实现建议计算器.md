#### v-model: 是唯一一个数据双向绑定指令

```
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <meta http-equiv="X-UA-Compatible" content="ie=edge">
    <title>Document</title>
    <script src="./lib/vue.js"></script>
</head>
<body>
    <div id="app">
        <h4>{{ msg }}</h4>
        <!-- v-bind:只能实现数据的单向绑定, 从M 自动绑定到V, 无法实现数据的双向绑定. 在页面上修改input的内容, msg不会修改 -->
        <input type="text" :value="msg" style="width:100%">
        <br data-tomark-pass>        <!-- 使用 v-model 指令，可以实现表单元素和Model 中数据的双向绑定 -->
        <!-- 注意：v-model 只能运用在表单元素中(input(radio,text,address...), select, checkbox, textarea) -->
        <input type="text" style="width:100%" v-model="msg">
    </div>
    <script>
        // 控制台 vm.msg="123"，可以修改页面数据，只是单向，不是双向
        var vm  = new Vue({
            el: '#app',
            data: {
                msg: 'hello'
            },
            methods: {
            }
        });
    </script>
</body>
</html>

```

            el: '#app',

            },

                calc() {

                    var codeStr = 'parseFloat(this.num1)' + this.opt + 'parseFloat(this.num2)';

                }
            }

    </script>
</body>
</html>`
