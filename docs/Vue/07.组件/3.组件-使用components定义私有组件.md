#### 通过components 定义私有组件

```
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <meta http-equiv="X-UA-Compatible" content="ie=edge">
    <title>Document</title>
    <script src="./lib/vue-2.4.0.js"></script>
</head>
<body>
    <div id="app">
        <login></login>
    </div>
    <template id="tmpl">
        <div>
            <h1>这是私有的login 组件</h1>
        </div>
    </template>
    <script>
        var vm = new Vue({
            el: '#app',
            data: {},
            methods: {},
            components: {       // 定义实例内部私有组建
                login: {
                    //template: '<h1>这是私有的login 组件</h1>'
                    template: '#tmpl'
                }
            }
        });
    </script>
</body>
</html>

```
