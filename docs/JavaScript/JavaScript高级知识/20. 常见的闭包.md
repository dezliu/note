1.
将函数作为另外一个函数的返回值

1.
将函数作为实参传递给另一个函数调用

**eg:**

```
// 1. 将函数作为另外一个函数的返回值
function fn1() {
    var a = 2;
    function fn2() {
        a++;
        console.log(a);
    }
    return fn2;
}
var f = fn1();
f()         // 3
f()         // 4

// 2. 将函数作为实参传递给另一个函数调用
function showDelay(msg, time) {
    setTimeout(function() {
        alert(msg);                         // msg 是闭包变量，time 不是
    }, time);
}
showDelay("ldzlxs", 2000);

```
