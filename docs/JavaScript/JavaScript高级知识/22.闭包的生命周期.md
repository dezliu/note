# 22.闭包的生命周期

- 笔记本：JavaScript高级知识
- 创建时间：2020-01-14 02:13:36 UTC
- 更新时间：2020-01-14 02:18:42 UTC
- 印象笔记 GUID：c95cdfa7-a0de-4afb-ba1f-80299cd097ff

1. 产生：在嵌套内部函数定义执行完时就产生了（不是在调用）

1. 死亡：在嵌套的内部函数成为垃圾对象时

```
function fn1() {
    // 闭包此时就产生了（函数提升，内部函数对象已经创建了）
    var a = 2;
    function fn2() {
        a++;
        console.log(a);
    }
    return fn2;
}
var f = fn1();      // 执行此行，闭包就已经存在了
f()         // 3
f()         // 4
f = null;           // 闭包死亡（包含闭包的函数对象成为了垃圾对象）

```

```
function fn1() {
    var a = 2;
    var fn2 = function() {  // 函数定义执行完才产生闭包
        a++;
        console.log(a);
    }
    return fn2;
}
var f = fn1();
f()         // 3
f()         // 4
f = null;           // 闭包死亡（包含闭包的函数对象成为了垃圾对象）

```
