#### 闭包缺点及解决

1. 缺点

- 函数执行完后，函数内的局部变量没有释放，内存占用时间会边长

- 容易照成内存泄漏

1. 解决

- 能不用闭包就不用

- 及时释放

```
<script>
    function fn1() {
        var arr = new Array[100000];
        function fn2() {
            console.log(arr.length);
        }
        return fn2;
    }
    var f = fn1();
    f();

    f = null;           // 让内部函数成为垃圾对象 --> 回收闭包
</script>

```

#### 内存溢出及内存泄漏

1. 内存溢出

- 一种程序运行出现的错误

- 当程序运行需要的内存超过了剩余的内存时，就会抛出内存溢出的错误

1. 内存泄漏

- 占用的内存没有及时释放

- 内存泄漏积累多了就会导致内存溢出

- 常见的内存泄漏

  - 意外的全局变量

  - 没有及时清理的计时器或回调函数

  - 闭包

```
<script>
    // 1. 内存溢出
    var obj = {};
    for(var i = 0; i < 10000; i++) {
        obj[i] = new Array(10000000);
    }

    // 2. 内存泄漏
        // 意外的局部变量
    function fn() {
        a = 3;              // 没有用 var 定义，是全局变量
        console.log(a);
    }
    fn();

    // 没有及时清理的计时器或回调函数
    var intervalId = setInterval(function() {        // 启动循环定时器后不清理，
        console.log("-----");
    }, 1000);

    clearInterval(intervalId);                          // 清楚循环定时器

    // 闭包
    function fn1() {
        var a = 4;
        function fn2() {
            console.log(++a);
        }
        return fn2;
    }

    var f = fn1();
    f();

    f = null;
</script>

```
