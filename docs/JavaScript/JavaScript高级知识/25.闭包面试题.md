#### 面试题1

```
<script>
    // 代码片段一
    var name = "The Window";
    var object = {
        name: "myObject",
        getNameFunc: function() {
            return function() {
                return this.name;               // 没有闭包，内部函数没有引用外部函数的变量
            }
        }
    }
    alert(object.getNameFunc()());          // The window，全局的调用，this指向window

    // 代码片段二
    var name2 = "The Window";
    var object2 = {
        name2: "My Object",
        getNameFunc: function() {
            var self = this;
            return function() {
                return self.name2;          // 有闭包，内部函数引用了外部函数的 name2
            }
        }
    }
    alert(object2.getNameFunc()());     // My Object
</script>

```

#### 面试题2

```
<script>
    function fun(n, o) {
        console.log(o);
        return {
            fun: function(m) {
                return fun(m,n);
            }
        }
    }
    var a = fun(0); // undefine
    a.fun(1);         // 0
    a.fun(2);         // 0
    a.fun(3);         // 0
    var b = fun(0).fun(1).fun(2).fun(3);    // undefined 0 1 2
    var c = fun(0).fun(1);                  // undefined 0
    c.fun(2);                                   // 1
    c.fun(3);                                   // 1
</script>

```
